So far we studied unconstrained problems, where a local minimum must be a stationary point. Now we move to problems with constraints: the condition $\nabla f(x^*) = 0$ is no longer necessary, and we need optimality conditions that also take the constraints into account.

\subsection{First-order optimality conditions for constrained problems}

Consider the constrained optimization problem
\[
    \min f(x), \quad x \in X \coloneqq \{x \in \R^n : g_i(x) \le 0,\ i = 1, \dots, m,\ h_j(x) = 0,\ j = 1, \dots, p\}, \tag{P}
\]
where $f$, $g_i$ and $h_j$ are continuously differentiable for any $i, j$. We write $g = (g_1, \dots, g_m)$ and $h = (h_1, \dots, h_p)$, so the constraints are $g(x) \le 0$ and $h(x) = 0$.

To describe the feasible set around a point $x^*$ we use two cones: one built from the geometry of $X$ and one built from the gradients of the constraints.

\begin{definition}[Tangent cone]
    The \g{tangent cone} at $x^* \in X$ is
    \[
        T_X(x^*) = \left\{ d \in \R^n : \exists\, \{z_k\} \subset X,\ \exists\, \{t_k\} > 0,\ z_k \to x^*,\ t_k \to 0,\ \lim_{k \to \infty} \frac{z_k - x^*}{t_k} = d \right\}.
    \]
\end{definition}

So $T_X(x^*)$ contains the directions along which we can approach $x^*$ while staying in $X$.

\begin{definition}[Active constraints]
    The set of \g{active constraints} at $x^* \in X$ is
    \[
        A(x^*) = \{i : g_i(x^*) = 0\},
    \]
    i.e., the inequality constraints satisfied with equality at $x^*$.
\end{definition}

\begin{definition}[First-order feasible direction cone]
    The \g{first-order feasible direction cone} at $x^* \in X$ is
    \[
        D(x^*) = \left\{ d \in \R^n :
        \begin{aligned}
            & d^T\nabla g_i(x^*) \le 0 && \forall\, i \in A(x^*),\\
            & d^T\nabla h_j(x^*) = 0 && \forall\, j = 1, \dots, p
        \end{aligned}
        \right\}.
    \]
\end{definition}

$D(x^*)$ is the linearization of the feasible set at $x^*$: only the active constraints matter, because the inactive ones remain satisfied in a neighborhood of $x^*$.

\subsection{Abadie constraint qualification (ACQ)}

The tangent cone is the "true" local description of $X$, but it is hard to compute. The cone $D(x^*)$ is easy to compute, since it only needs gradients. We would like the two cones to coincide.

\begin{definition}[Abadie constraint qualification]
    We say that the \g{Abadie constraint qualification} (ACQ) holds at a point $x^* \in X$ if
    \[
        T_X(x^*) = D(x^*).
    \]
\end{definition}

Checking ACQ directly from the definition is not easy either, so the following theorem gives some sufficient conditions that are simple to verify.

\begin{theorem}[Sufficient conditions for ACQ]
    \begin{enumerate}[label=\alph*)]
        \item \g{Affine constraints}: if $g_i$ and $h_j$ are affine for all $i = 1, \dots, m$ and $j = 1, \dots, p$, then ACQ holds at any $x \in X$.
        \item \g{Slater condition} for convex problems: if $g_i$ are convex for all $i = 1, \dots, m$, $h_j$ are affine for all $j = 1, \dots, p$ and there exists $\bar{x} \in X$ s.t.\ $g(\bar{x}) < 0$ and $h(\bar{x}) = 0$, then ACQ holds at any $x \in X$.
        \item \g{Linear independence} of the gradients of active constraints: if $x^* \in X$ and the vectors
        \[
            \nabla g_i(x^*) \text{ for } i \in A(x^*), \qquad \nabla h_j(x^*) \text{ for } j = 1, \dots, p
        \]
        are linearly independent, then ACQ holds at $x^*$.
    \end{enumerate}
\end{theorem}

\subsection{Karush-Kuhn-Tucker theorem}

The first step is a geometric necessary condition: at a local minimum, no direction that stays in $X$ can be a descent direction.

\begin{lemma}
    If $x^*$ is a local minimum of (P), then
    \[
        d^T\nabla f(x^*) \ge 0 \qquad \forall\, d \in T_X(x^*).
    \]
\end{lemma}

This condition uses $T_X(x^*)$, which is hard to handle. If ACQ holds, we can replace $T_X(x^*)$ with $D(x^*)$, and this leads to an algebraic system in terms of the gradients.

\begin{theorem}[Karush-Kuhn-Tucker]
    \label{thm:kkt}
    If $x^*$ is a local minimum and ACQ holds at $x^*$, then there exist $\lambda^* \in \R^m$ and $\mu^* \in \R^p$ s.t.\ $(x^*, \lambda^*, \mu^*)$ satisfies the \g{KKT system}:
    \[
        \begin{cases}
            \nabla f(x^*) + \sum_{i=1}^{m} \lambda_i^* \nabla g_i(x^*) + \sum_{j=1}^{p} \mu_j^* \nabla h_j(x^*) = 0,\\
            \lambda_i^* g_i(x^*) = 0 \quad \forall\, i = 1, \dots, m,\\
            \lambda^* \ge 0,\\
            g(x^*) \le 0,\\
            h(x^*) = 0.
        \end{cases}
    \]
\end{theorem}

The vectors $\lambda^*$ and $\mu^*$ are called \g{KKT multipliers}. The KKT system can be written in a compact form using the following function.

\begin{definition}[Lagrangian function]
    The \g{Lagrangian function} $L : \R^n \times \R^m \times \R^p \to \R$ is
    \[
        L(x, \lambda, \mu) \coloneqq f(x) + \sum_{i=1}^{m} \lambda_i g_i(x) + \sum_{j=1}^{p} \mu_j h_j(x).
    \]
\end{definition}

Then the KKT system can be formulated as:
\[
    \begin{cases}
        \nabla_x L(x, \lambda, \mu) = 0,\\
        \lambda_i g_i(x) = 0, \quad i = 1, \dots, m,\\
        \lambda \ge 0,\ h(x) = 0,\ g(x) \le 0.
    \end{cases}
\]

\begin{remark}
    Since $\lambda \ge 0$ and $g(x) \le 0$, every term $\lambda_i g_i(x)$ is $\le 0$. So the condition $\lambda_i g_i(x) = 0$ for $i = 1, \dots, m$ is equivalent to $\lambda^T g(x) = 0$, or also $\langle \lambda, g(x) \rangle = 0$ (\g{complementarity condition}).
\end{remark}

\begin{example}
    \label{ex:kkt-circle}
    Find the minimum points of the function $f(x_1, x_2) = 2x_1 + x_2$ on the set
    \[
        X = \{(x_1, x_2) \in \R^2 : x_1^2 + x_2^2 \le 4\}.
    \]
    Note that $X$ is compact, so from the Weierstrass theorem $f$ admits a global maximum and a global minimum on $X$.

    The Lagrangian function is
    \[
        L(x_1, x_2, \lambda) = 2x_1 + x_2 + \lambda(x_1^2 + x_2^2 - 4).
    \]
    The KKT system is given by:
    \[
        \begin{cases}
            2 + 2\lambda x_1 = 0,\\
            1 + 2\lambda x_2 = 0,\\
            \lambda(x_1^2 + x_2^2 - 4) = 0,\\
            x_1^2 + x_2^2 \le 4,\\
            \lambda \ge 0.
        \end{cases}
    \]
    Note that for $\lambda = 0$ the system is impossible (the first equation becomes $2 = 0$). So $\lambda > 0$, the constraint is active, and we are led to solve the system:
    \[
        \begin{cases}
            x_1 = -\dfrac{1}{\lambda},\\[2mm]
            x_2 = -\dfrac{1}{2\lambda},\\[2mm]
            x_1^2 + x_2^2 = 4,\\
            \lambda \ge 0.
        \end{cases}
    \]
    Then
    \[
        \frac{1}{\lambda^2} + \frac{1}{4\lambda^2} = 4,
    \]
    from which
    \[
        16\lambda^2 = 5, \qquad \lambda = \pm \frac{\sqrt{5}}{4}.
    \]
    Since $\lambda \ge 0$, we only keep $\lambda = \frac{\sqrt{5}}{4}$:
    \[
        x_1 = -\frac{4}{\sqrt{5}} = -\frac{4\sqrt{5}}{5}, \qquad x_2 = -\frac{2}{\sqrt{5}} = -\frac{2\sqrt{5}}{5}.
    \]
    The global minimum exists and, as shown in the remark below, ACQ holds at every point of $X$, so by \cref{thm:kkt} the global minimum must solve the KKT system. Since the KKT system has only one solution, it follows that $\bar{x} = \left(-\frac{4\sqrt{5}}{5}, -\frac{2\sqrt{5}}{5}\right)$ is the global minimum point.
\end{example}

\begin{remark}
    In the previous example ACQ is fulfilled for every $x \in X$. Indeed, there is only the constraint $g(x) = x_1^2 + x_2^2 - 4 \le 0$, with $\nabla g(x_1, x_2) \neq (0, 0)$ for every $x \in X$ s.t.\ $x_1^2 + x_2^2 - 4 = 0$. So condition c) of the sufficient conditions for ACQ holds.
\end{remark}

What about the maximum points of $f$ on $X$? It is enough to set $\lambda \le 0$ in the KKT system. In fact, to find the maxima of $f$ we look for the minima of $-f$. The KKT system for the problem
\[
    \min\, (-f(x)), \quad x \in X
\]
is
\[
    \begin{cases}
        -2 + 2\lambda x_1 = 0,\\
        -1 + 2\lambda x_2 = 0,\\
        \lambda(x_1^2 + x_2^2 - 4) = 0,\\
        x_1^2 + x_2^2 \le 4,\\
        \lambda \ge 0,
    \end{cases}
\]
which is equivalent to setting $\lambda \le 0$ in the original one. Choosing $\lambda = -\frac{\sqrt{5}}{4}$ in the original system we obtain
\[
    x_1 = \frac{4}{\sqrt{5}} = \frac{4\sqrt{5}}{5}, \qquad x_2 = \frac{2}{\sqrt{5}} = \frac{2\sqrt{5}}{5},
\]
so that $\hat{x} = \left(\frac{4\sqrt{5}}{5}, \frac{2\sqrt{5}}{5}\right)$ is the global maximum point of $f$ on $X$.

\subsubsection{Limits of the KKT theorem}

The KKT theorem has two limits: it needs ACQ, and it only gives necessary conditions.

\begin{remark}
    The ACQ assumption is crucial in the KKT theorem.
\end{remark}

\begin{example}
    Consider
    \[
        \begin{cases}
            \min\, x_1 + x_2\\
            (x_1 - 1)^2 + (x_2 - 1)^2 - 1 \le 0\\
            x_2 \le 0
        \end{cases}
    \]
    The feasible set is the single point $x^* = (1, 0)$, which is therefore the global optimum.

    $T_X(x^*) = \{0\}$ and $D(x^*) = \{d \in \R^2 : d_2 = 0\}$, hence ACQ does not hold at $x^*$.

    $\nabla g_1(x^*) = (0, -2)$, $\nabla g_2(x^*) = (0, 1)$, $\nabla f(x^*) = (1, 1)$, hence there is no $\lambda^*$ s.t.\ $(x^*, \lambda^*)$ solves the KKT system: the first component of $\nabla_x L$ is always $1 \neq 0$.
\end{example}

The KKT theorem gives necessary optimality conditions, but not sufficient ones.

\begin{example}
    Consider
    \[
        \begin{cases}
            \min\, x_1^2 - x_2^2\\
            x_1^2 + x_2^2 - 2 \le 0
        \end{cases}
    \]
    $x^* = (0, 0)$, $\lambda^* = 0$ solves the KKT system, but $x^*$ is not a local optimum.
\end{example}

For convex problems, instead, the KKT conditions are also sufficient.

\begin{theorem}
    \label{thm:kkt-convex}
    If the optimization problem is convex and $(x^*, \lambda^*, \mu^*)$ solves the KKT system, then $x^*$ is a global optimum.
\end{theorem}

Recall that (P) is convex if $f$ and $g$ are convex and $h$ is affine.

\subsection{Lagrangian duality}

\subsubsection{Lagrangian relaxation and dual function}

The Lagrangian function is also the starting point of duality theory. The idea is to remove the constraints and move them into the objective function, weighted by the multipliers. Denote by $v(P)$ the optimal value of (P).

\begin{definition}[Lagrangian relaxation]
    Given $\lambda \ge 0$ and $\mu \in \R^p$, the problem
    \[
        \inf L(x, \lambda, \mu), \quad x \in \R^n
    \]
    is called \g{Lagrangian relaxation} of (P), and
    \[
        \varphi(\lambda, \mu) = \inf_{x \in \R^n} L(x, \lambda, \mu)
    \]
    is the \g{Lagrangian dual function}.
\end{definition}

The Lagrangian relaxation provides a lower bound to $v(P)$.

\begin{theorem}
    \label{thm:lower-bound}
    For any $\lambda \ge 0$ and $\mu \in \R^p$, we have $\varphi(\lambda, \mu) \le v(P)$.
\end{theorem}

\begin{proof}
    If $x \in X$, i.e., $g(x) \le 0$ and $h(x) = 0$, then
    \[
        L(x, \lambda, \mu) = f(x) + \sum_{i=1}^{m} \lambda_i g_i(x) + \sum_{j=1}^{p} \mu_j h_j(x) \le f(x),
    \]
    hence
    \[
        \varphi(\lambda, \mu) = \inf_{x \in \R^n} L(x, \lambda, \mu) \le \inf_{x \in X} L(x, \lambda, \mu) \le \inf_{x \in X} f(x) = v(P). \qedhere
    \]
\end{proof}

The dual function $\varphi$ has the following properties:
\begin{itemize}
    \item it is concave, because it is the infimum of affine functions w.r.t.\ $(\lambda, \mu)$;
    \item it may be equal to $-\infty$ at some point;
    \item it may be not differentiable at some point.
\end{itemize}

\subsubsection{Lagrangian dual problem}

Each choice of $(\lambda, \mu)$ gives a lower bound, so it is natural to look for the best one.

\begin{definition}[Lagrangian dual problem]
    The problem
    \[
        \max \varphi(\lambda, \mu), \quad \lambda \ge 0,\ \mu \in \R^p \tag{D}
    \]
    is called \g{Lagrangian dual problem} of (P), and (P) is called \g{primal problem}.
\end{definition}

\begin{itemize}
    \item The dual problem (D) consists in finding the best lower bound of $v(P)$.
    \item (D) is always equivalent to a convex problem, even if (P) is non-convex: indeed, it is the maximization of a concave function on a convex set.
\end{itemize}

\Cref{thm:lower-bound} can be equivalently stated as follows.

\begin{theorem}[Weak duality]
    For any optimization problem (P), we have $v(D) \le v(P)$.
\end{theorem}

The previous inequality is called \g{weak duality property}.

\begin{example}[Linear programming]
    Primal problem:
    \[
        \begin{cases}
            \min\, c^Tx\\
            Ax \ge b
        \end{cases} \tag{P}
    \]
    Lagrangian function: $L(x, \lambda) = c^Tx + \lambda^T(b - Ax) = \lambda^Tb + (c^T - \lambda^TA)x$.

    Dual function:
    \[
        \varphi(\lambda) = \inf_{x \in \R^n} L(x, \lambda) =
        \begin{cases}
            -\infty & \text{if } c^T - \lambda^TA \neq 0,\\
            \lambda^Tb & \text{if } c^T - \lambda^TA = 0.
        \end{cases}
    \]
    Dual problem:
    \[
        \begin{cases}
            \max\, \varphi(\lambda)\\
            \lambda \ge 0
        \end{cases}
        \quad \longrightarrow \quad
        \begin{cases}
            \max\, \lambda^Tb\\
            \lambda^TA = c^T\\
            \lambda \ge 0
        \end{cases} \tag{D}
    \]
    which is a linear programming problem.
\end{example}

\begin{example}[Least norm solution of linear equations]
    Primal problem:
    \[
        \begin{cases}
            \min\, \frac{1}{2}x^Tx\\
            Ax = b
        \end{cases} \tag{P}
    \]
    Lagrangian function: $L(x, \mu) = \frac{1}{2}x^Tx + \mu^T(b - Ax)$.

    Dual function: $\varphi(\mu) = \inf_{x \in \R^n} L(x, \mu)$.

    $L(x, \mu)$ is quadratic and strongly convex with respect to $x$, therefore
    \[
        \varphi(\mu) = \inf_{x \in \R^n} L(x, \mu) = \min_{x \in \R^n} L(x, \mu),
    \]
    thus the global optimum is the stationary point:
    \[
        \nabla_x L = x - A^T\mu = 0 \iff x = A^T\mu,
    \]
    hence $\varphi(\mu) = -\frac{1}{2}\mu^TAA^T\mu + b^T\mu$.

    Dual problem:
    \[
        \begin{cases}
            \max\, -\frac{1}{2}\mu^TAA^T\mu + b^T\mu\\
            \mu \in \R^p
        \end{cases} \tag{D}
    \]
    which is an unconstrained convex quadratic programming problem.
\end{example}

\subsubsection{Strong duality}

Weak duality says that $v(D) \le v(P)$. The interesting case is when the two values coincide.

\begin{definition}[Strong duality]
    We say that \g{strong duality} holds for (P) if $v(D) = v(P)$ and (D) admits an optimal solution.
\end{definition}

Strong duality does not hold in general.

\begin{example}
    Consider the following (non-convex) problem:
    \[
        \begin{cases}
            \min\, -x^2\\
            x - 1 \le 0\\
            -x \le 0
        \end{cases} \tag{P}
    \]
    It is easy to check that $v(P) = -1$. The Lagrangian function is $L(x, \lambda) = -x^2 + \lambda_1(x - 1) - \lambda_2x$, hence
    \[
        \varphi(\lambda) = \inf_{x \in \R} L(x, \lambda) = -\infty \qquad \forall\, (\lambda_1, \lambda_2) \in \R^2,
    \]
    so $v(D) = -\infty$.
\end{example}

The next theorem provides sufficient conditions which guarantee strong duality for (P).

\begin{theorem}
    Suppose $f$, $g$, $h$ are continuously differentiable, the primal problem
    \[
        \begin{cases}
            \min\, f(x)\\
            g(x) \le 0\\
            h(x) = 0
        \end{cases} \tag{P}
    \]
    is convex, there exists a global optimum $x^*$ and ACQ holds at $x^*$. Then:
    \begin{itemize}
        \item strong duality holds ($v(D) = v(P)$ and (D) admits an optimal solution);
        \item $(\lambda^*, \mu^*)$ is optimal for (D) if and only if $(\lambda^*, \mu^*)$ is a KKT multipliers vector associated with $x^*$.
    \end{itemize}
\end{theorem}

\begin{proof}
    $L(x, \lambda, \mu)$ is convex with respect to $x$ since (P) is convex. Since ACQ holds at $x^*$, by \cref{thm:kkt} there exists a KKT vector of multipliers associated with $x^*$. Let $(\lambda^*, \mu^*)$ be any such vector. Then
    \[
        \nabla_x L(x^*, \lambda^*, \mu^*) = 0, \qquad \lambda^* \ge 0, \qquad (\lambda^*)^Tg(x^*) = 0.
    \]
    Thus, using that $L$ is convex in $x$ (so a stationary point is a global minimum) and weak duality,
    \[
        \begin{aligned}
            v(D) \ge \varphi(\lambda^*, \mu^*) &= \inf_x L(x, \lambda^*, \mu^*) = L(x^*, \lambda^*, \mu^*)\\
            &= f(x^*) + (\lambda^*)^Tg(x^*) + (\mu^*)^Th(x^*) = f(x^*) = v(P) \ge v(D).
        \end{aligned}
    \]
    Therefore $v(P) = v(D)$ and $(\lambda^*, \mu^*)$ is optimal for (D).

    Vice versa, if $(\lambda^*, \mu^*)$ is any optimal solution for (D), then
    \[
        \begin{aligned}
            f(x^*) &= v(P) = v(D) = \varphi(\lambda^*, \mu^*) = \inf_x L(x, \lambda^*, \mu^*) \le L(x^*, \lambda^*, \mu^*)\\
            &= f(x^*) + (\lambda^*)^Tg(x^*) + (\mu^*)^Th(x^*)\\
            &= f(x^*) + (\lambda^*)^Tg(x^*) \le f(x^*),
        \end{aligned}
    \]
    thus $(\lambda^*)^Tg(x^*) = 0$ and $\inf_x L(x, \lambda^*, \mu^*) = L(x^*, \lambda^*, \mu^*)$, hence $\nabla_x L(x^*, \lambda^*, \mu^*) = 0$, i.e., $(\lambda^*, \mu^*)$ is a KKT multipliers vector associated with $x^*$.
\end{proof}

Convexity is sufficient but not necessary: strong duality may hold also for some non-convex problems.

\begin{example}
    Consider the (non-convex) problem
    \[
        \begin{cases}
            \min\, -x_1^2 - x_2^2\\
            x_1^2 + x_2^2 - 1 \le 0
        \end{cases}
    \]
    We have $v(P) = -1$. The Lagrangian function is
    \[
        L(x, \lambda) = -x_1^2 - x_2^2 + \lambda(x_1^2 + x_2^2 - 1) = (\lambda - 1)x_1^2 + (\lambda - 1)x_2^2 - \lambda.
    \]
    The dual function is
    \[
        \varphi(\lambda) = \inf_{x \in \R^2} L(x, \lambda) =
        \begin{cases}
            -\infty & \text{if } \lambda < 1,\\
            -\lambda & \text{if } \lambda \ge 1.
        \end{cases}
    \]
    The dual problem is
    \[
        \begin{cases}
            \max\, -\lambda\\
            \lambda \ge 1
        \end{cases}
    \]
    hence its optimal solution is $\lambda^* = 1$ and $v(D) = -1 = v(P)$.
\end{example}

Finally, strong duality can be characterized through the saddle points of the Lagrangian function.

\begin{theorem}[Characterization of strong duality]
    $(x^*, \lambda^*, \mu^*)$ is a \g{saddle point} of $L$, i.e.,
    \[
        L(x^*, \lambda, \mu) \le L(x^*, \lambda^*, \mu^*) \le L(x, \lambda^*, \mu^*) \qquad \forall\, x \in \R^n,\ \lambda \in \R^m_+,\ \mu \in \R^p,
    \]
    if and only if $x^*$ is optimal for (P), $(\lambda^*, \mu^*)$ is optimal for (D) and $v(P) = v(D)$.
\end{theorem}

\begin{proof}
    If $(x^*, \lambda^*, \mu^*)$ is a saddle point of $L$, then we can prove that $x^* \in X$ and $\langle \lambda^*, g(x^*) \rangle = 0$, which implies $\varphi(\lambda^*, \mu^*) = f(x^*)$.

    Vice versa, we have that
    \[
        f(x^*) = \varphi(\lambda^*, \mu^*) = \inf_{x \in \R^n} L(x, \lambda^*, \mu^*) \le L(x^*, \lambda^*, \mu^*) = f(x^*) + \langle \lambda^*, g(x^*) \rangle,
    \]
    hence $\langle \lambda^*, g(x^*) \rangle = 0$ and
    \[
        L(x^*, \lambda^*, \mu^*) = f(x^*) = \varphi(\lambda^*, \mu^*) \le L(x, \lambda^*, \mu^*) \qquad \forall\, x \in \R^n.
    \]
    Moreover
    \[
        L(x^*, \lambda, \mu) = f(x^*) + \langle \lambda, g(x^*) \rangle \le f(x^*) = L(x^*, \lambda^*, \mu^*)
    \]
    for all $\lambda \in \R^m_+$ and $\mu \in \R^p$.
\end{proof}
