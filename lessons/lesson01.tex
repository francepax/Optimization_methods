\section{Lesson 01 --- 24/09/2026}

\subsection{Introduction to optimization problems}

The aim of the course is to give an overview, from the mathematical point of view, of the main optimization methods for data analysis and decision problems. The main tool is an optimization problem
\[
    \min \{ f(x) : x \in X \}, \tag{P}
\]
where $f: \R^n \to \R$ is the \g{objective function} and $X \subseteq \R^n$ is the \g{constraints set} (or \g{feasible region}). If $X = \R^n$ the problem is said to be \g{unconstrained}.

\begin{remark}
(P) can be seen as a decision problem: $X$ is the set of all the admissible (feasible) decisions $x$ and $f(x)$ is the value of the decision $x$ (for example a cost or a gain).
\end{remark}

In general $X$ is described by constraint functions,
\[
    \begin{cases}
        \min f(x)\\
        g(x) \le 0\\
        h(x) = 0
    \end{cases}
\]
where $g = (g_1, \dots, g_m)$ are the inequality constraints functions and $h = (h_1, \dots, h_p)$ are the equality constraints functions. We will generally consider minimization problems, since $\max\{f(x) : x \in X\} = -\min\{-f(x) : x \in X\}$.

It is fundamental to analyze the properties of (P) under suitable assumptions on $X$ and on the involved functions, mainly \g{convexity} and \g{differentiability}. This is why we start from some notions of convex analysis.

\subsubsection{An application: Economical Power Dispatch}

A power system has to satisfy a given electricity load demand $P_D$. The electricity is produced by $n$ different units, each with a fuel cost function. The \g{Economical Power Dispatch} (EPD) consists in determining the load to be produced by each unit, in a given interval of time $T$, in order to minimize the total fuel cost while supplying the total demand and respecting some technical requirements.

The model is the following:
\begin{itemize}
    \item $p_i$ (in MW) is the unknown power generated by unit $i = 1, \dots, n$;
    \item $f_i(p_i) = a_i p_i^2 + b_i p_i + c_i$ is the fuel cost (Euro/T) of unit $i$, with $a_i, b_i, c_i$ the fuel cost coefficients;
    \item $p_i^{\min}$ and $p_i^{\max}$ are the minimum and maximum output of unit $i$;
    \item $P_D$ is the total demand and $P_L$ the system losses (in MW).
\end{itemize}
\[
    \begin{cases}
        \min F(p) \coloneqq \sum_{i=1}^{n} f_i(p_i)\\
        p_1 + p_2 + \dots + p_n = P_D + P_L\\
        p_i^{\min} \le p_i \le p_i^{\max}, \quad i = 1, \dots, n
    \end{cases}
\]

\begin{remark}
If we also want to minimize the emissions $E(p) = \sum_i e_i(p_i)$, with $e_i(p_i) = \alpha_i p_i^2 + \beta_i p_i + \gamma_i$, we get a \g{multiobjective} problem $\min (F(p), E(p))$ with the same constraints (Environmental/Economical Power Dispatch).
\end{remark}

The course program covers: convex analysis, existence of optima, optimality conditions and duality, solution methods (gradient, conjugate gradient, Newton), applications to machine learning (SVM), multiobjective optimization and non-cooperative game theory.

\subsection{Convex sets}

\begin{definition}[Convex set]
A set $C \subseteq \R^n$ is \emph{convex} if, for every $x, y \in C$ and every $\alpha \in [0,1]$,
\[
  \alpha x + (1-\alpha) y \in C.
\]
\end{definition}

In words, the segment joining any two points of $C$ is contained in $C$.

\begin{definition}[Affine set]
A set $C \subseteq \R^n$ is \emph{affine} if, for every $x, y \in C$ and every $\alpha \in \R$,
\[
  \alpha x + (1-\alpha) y \in C.
\]
\end{definition}

\begin{remark}
Every affine set is convex, since $[0,1] \subset \R$.
\end{remark}

Examples of affine sets are: any single point $\{x\}$, any line, any subspace, the solution set of a linear system $C = \{x \in \R^n : Ax = b\}$, with $A$ an $m \times n$ matrix and $b \in \R^m$.

A \emph{subspace} is a particular affine set: $S \subseteq \R^n$ is a subspace if $\alpha x + \beta y \in S$ for every $x, y \in S$ and $\alpha, \beta \in \R$. Examples are $\{0\}$, any line through zero, the solution set of a homogeneous system $\{x : Ax = 0\}$.

\begin{definition}[Convex combination]
A \emph{convex combination} of the points $x^1,x^2,\dots,x^k$ is a point
\[
    y=\sum_{i=1}^{k} \alpha_i x^i
    \quad \text{where} \quad
    \alpha_1,\dots,\alpha_k \in [0,1]
    \quad\text{and} \quad
     \sum_{i=1}^{k}\alpha_i=1.
\]
\end{definition}

\begin{remark}
By definition, a set $C \subseteq \R^n$ is \emph{convex} if it contains all the convex combinations of any two points in $C$.
\end{remark}

\begin{example}
Take $x^1 = (1,1)$, $x^2 = (3,1)$, $x^3 = (2,3)$. The point $x = (2,2)$ is a convex combination of them:
\[
    x = \tfrac{1}{4} x^1 + \tfrac{1}{4} x^2 + \tfrac{1}{2} x^3.
\]
\end{example}

The next lemma says that convexity is enough to be closed under convex combinations of \emph{any} number of points, not only two.

\begin{lemma}
If $C$ is convex, then for any $x^1,\dots,x^k \in C$ and $\alpha_1,\dots,\alpha_k \in [0,1]$ such that $\sum_{i=1}^{k}\alpha_i=1$,
\[
    \sum_{i=1}^{k}\alpha_ix^i \in C.
\]
\end{lemma}

\begin{proof}
By induction on $k$. For $k = 2$ the thesis holds by definition of convexity. Assume it holds for $k$ and let $x^1, \dots, x^{k+1} \in C$, $\alpha_1, \dots, \alpha_{k+1} \in [0,1]$ with $\sum_{i=1}^{k+1} \alpha_i = 1$. Without loss of generality $\alpha_1 \neq 1$. Then
\[
    \sum_{i=1}^{k+1} \alpha_i x^i = \alpha_1 x^1 + \Big(\sum_{i=2}^{k+1} \alpha_i\Big) \sum_{i=2}^{k+1} \frac{\alpha_i}{\sum_{j=2}^{k+1} \alpha_j}\, x^i .
\]
Since $\sum_{i=2}^{k+1} \frac{\alpha_i}{\sum_{j=2}^{k+1} \alpha_j} = 1$, by the inductive assumption $\bar{x} \coloneqq \sum_{i=2}^{k+1} \frac{\alpha_i}{\sum_{j=2}^{k+1} \alpha_j} x^i \in C$. Finally, since $C$ is convex and $\sum_{i=2}^{k+1}\alpha_i = 1 - \alpha_1$, the point $\alpha_1 x^1 + (1-\alpha_1)\bar{x}$ belongs to $C$.
\end{proof}

\begin{proposition}
If $\{C_i\}_{i \in I}$ is any (possibly infinite) family of convex sets, then $\bigcap_{i \in I} C_i$ is convex.
\end{proposition}

This proposition makes the following definition well posed.

\begin{definition}[Convex hull]
The \emph{convex hull} $\mathrm{conv}(C)$ of a set $C$ is the intersection of all the convex sets containing $C$. In other words, it is the smallest convex set containing $C$.
\end{definition}

For example, the convex hull of $x^1, x^2, x^3$ above is the triangle with vertices the three points.

\begin{proposition}
$\mathrm{conv}(C) = \{ \text{all convex combinations of points in } C\}$.
\end{proposition}

\begin{proof}
The set of convex combinations of points in $C$ is a convex set containing $C$, hence $\mathrm{conv}(C)$ is contained in it. Conversely, $C \subseteq \mathrm{conv}(C)$ and $\mathrm{conv}(C)$ is convex, so by the previous lemma it contains every convex combination of points of $C$.
\end{proof}

\begin{remark}
$C$ is convex if and only if $C = \mathrm{conv}(C)$.
\end{remark}

\subsubsection{Examples of convex sets}

\begin{definition}[Polyhedron]
A \emph{polyhedron} $P$ is the intersection of a finite number of closed halfspaces in $\R^n$. A closed halfspace is the set of solutions of a linear inequality $a^Tx \le \beta$, with $a \in \R^n$, $\beta \in \R$. Hence
\[
    P = \{x \in \R^n : Ax \le b\},
\]
with $A$ an $m \times n$ matrix and $b \in \R^m$.
\end{definition}

A polyhedron is convex, since any closed halfspace is convex and the intersection of convex sets is convex.

\begin{definition}[Ball]
A \emph{ball} is $B(\bar{x}, r) \coloneqq \{z \in \R^n : \norm{z - \bar{x}} \le r\}$, where $\norm{\cdot}$ is any norm, e.g.
\begin{itemize}
    \item $\norm{x}_2 = \sqrt{\sum_i x_i^2}$ (Euclidean norm);
    \item $\norm{x}_1 = \sum_i |x_i|$ (Manhattan distance);
    \item $\norm{x}_\infty = \max_i |x_i|$ (Chebyshev norm);
    \item $\norm{x}_p = \big(\sum_i |x_i|^p\big)^{1/p}$, with $1 \le p < +\infty$;
    \item $\norm{x}_A = \sqrt{x^TAx}$, with $A$ symmetric and positive definite, i.e.\ $x^TAx > 0$ for all $x \neq 0$.
\end{itemize}
\end{definition}

Recall that a norm on a real vector space $X$ is a function $p: X \to \R$ such that $p(x+y) \le p(x) + p(y)$, $p(\alpha x) = |\alpha| p(x)$, and $p(x) = 0 \Leftrightarrow x = 0$; these imply $p(x) \ge 0$.

Balls are convex for any norm; we prove it in the next lesson, after some examples.
